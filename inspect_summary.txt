function buildArmySetup(a){
  ensureXp(a);
  const bytes=[];
  const u8=v=>bytes.push(v&255);
  const u16=v=>{u8(v);u8(v>>>8);};
  const u32=v=>{u16(v&0xFFFF);u16((v>>>16)&0xFFFF);};
  const s8=t=>{u16(t.length);for(let i=0;i<t.length;i++)u8(t.charCodeAt(i)&127);};
  u32(11);u32(0);u32(0);
  s8(EXPORT_FACT[a.faction]||"mk_fact_castile");
  u8(1);u16(0);
  u32(0);u8(1);u32(0);
  // general first: the unit whose codex class is 'com', else the first unit
  const idx=[...a.units.keys()];
  let gi=idx.find(i=>{const u=cxBy[a.units[i]];return u&&u.class==="com";});
  if(gi===undefined)gi=0;
  idx.splice(idx.indexOf(gi),1);idx.unshift(gi);
  u32(idx.length);
  for(const i of idx){
    s8(a.units[i]);
    u32(Math.max(0,Math.min(9,(a.xp||[])[i]||0)));
    u8(i===gi?1:0);u16(0);
  }
  for(let i=0;i<16;i++)u8(0);   // footer, all-zero in every sample
  return new Uint8Array(bytes);
}

  el.href=URL.createObjectURL(blob);
  el.download=((a.name||"host").replace(/[^\w\- ]+/g,"").trim()||"host")+".army_setup";
  el.click();

  toast("Setup downloaded — drop it in Attila's army_setups folder, then load it in the custom battle lobby");
};
